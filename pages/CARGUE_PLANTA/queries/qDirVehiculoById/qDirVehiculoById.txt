with contexto as (
  select
    nullif(
      nullif(
        '{{ appsmith.store.cp_update_context?.entity_id || "" }}',
        ''
      ),
      'null'
    )::uuid as vehiculo_id,

    nullif(
      nullif(
        '{{ appsmith.store.cp_update_context?.tenant_vehiculo_id || "" }}',
        ''
      ),
      'null'
    )::uuid as tenant_vehiculo_id,

    coalesce(
      nullif(
        nullif('{{ appsmith.store.cp_tenant_id || "" }}', ''),
        'null'
      )::uuid,

      nullif(
        nullif('{{ appsmith.store.tenant_id || "" }}', ''),
        'null'
      )::uuid,

      nullif(
        nullif(
          '{{ appsmith.store.current_user?.tenant_id || "" }}',
          ''
        ),
        'null'
      )::uuid
    ) as tenant_id
)

select
  v.id as vehiculo_id,
  tv.id as tenant_vehiculo_id,

  tv.tenant_id,

  v.codigo_interno,
  v.pais_placa,
  v.placa,

  v.tipo_vehiculo_terrestre_id,
  tvt.codigo_rndc as tipo_vehiculo_codigo,
  tvt.nombre as tipo_vehiculo,

  v.marca,
  v.modelo,
  v.color,
  v.capacidad_carga_kg,

  tv.transportadora_id,
  tv.tipo_vehiculo_id,

  tv.soat_no,
  tv.soat_vencimiento,

  tv.habilitado_nacional,
  tv.habilitado_internacional,
  tv.estado_administrativo,
  tv.observaciones,

  v.activo as identidad_activa,
  tv.activo as relacion_activa,

  case
    when v.activo = false
      then 'IDENTIDAD_INACTIVA'

    when tv.activo = false
      or tv.deleted_at is not null
      or tv.estado_administrativo <> 'ACTIVO'
      then 'VEHICULO_INACTIVO'

    when tv.soat_no is null
      or trim(tv.soat_no) = ''
      or tv.soat_vencimiento is null
      then 'SIN_SOAT'

    when tv.soat_vencimiento < current_date
      then 'SOAT_VENCIDO'

    when tv.soat_vencimiento <= current_date + 30
      then 'SOAT_POR_VENCER'

    else 'SOAT_VIGENTE'
  end as estado_documental,

  case
    when tv.soat_vencimiento is null
      then null
    else tv.soat_vencimiento - current_date
  end as dias_vigencia_soat,

  (
    v.activo = true
    and tv.activo = true
    and tv.deleted_at is null
    and tv.estado_administrativo = 'ACTIVO'
    and tv.habilitado_nacional = true
    and tv.soat_no is not null
    and trim(tv.soat_no) <> ''
    and tv.soat_vencimiento is not null
    and tv.soat_vencimiento >= current_date
  ) as disponible_para_viaje

from contexto c

join public.vehiculos v
  on v.id = c.vehiculo_id

join public.tenant_vehiculos tv
  on tv.id = c.tenant_vehiculo_id
 and tv.tenant_id = c.tenant_id
 and tv.vehiculo_id = v.id

left join public.tipos_vehiculo_terrestre tvt
  on tvt.id = v.tipo_vehiculo_terrestre_id

where c.vehiculo_id is not null
  and c.tenant_vehiculo_id is not null
  and c.tenant_id is not null
  and v.activo = true
  and tv.activo = true
  and tv.deleted_at is null

limit 1;
